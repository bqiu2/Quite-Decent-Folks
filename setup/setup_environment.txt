# setup_environment.ps1
# Create the shared Conda environment on Windows PowerShell.

# 1. Create the environment.
conda create -n plant-game python=3.12.13 -y

# 2. Activate the environment.
conda activate plant-game

# 3. Upgrade pip.
python -m pip install --upgrade pip

# 4. Remove conflicting OpenCV distributions.
python -m pip uninstall -y opencv-python opencv-python-headless opencv-contrib-python opencv-contrib-python-headless

# 5. Install the project dependencies.
python -m pip install -r requirements.txt

# 6. Check the environment.
python check_environment.py
